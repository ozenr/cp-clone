// Determine if File is Regular, Symbolic Link, or Directory
int file_type(int fd) {
	// Exit Status 
	int status = 0;

	// Create Stat Struct
	struct stat st; 

	// Grab Metadata Information
	if (fstat(fd, &st) == -1) {
		perror("stat");
		status = 1;
	}

	// Check If Regular File or Directory Using Flags
	switch(st.st_mode) {
		case S_IFREG:
			REG_F = 1;
			break;
		
		case S_IFDIR:
			DIR_F = 1;
			break;

		case S_IFLNK:
			LNK_F = 1;
			break;
	}
	
	return status;
}
